cmake_minimum_required(VERSION 3.18)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" REMOTEOS_VERSION LIMIT_COUNT 1)
project(RemoteOS VERSION ${REMOTEOS_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(REMOTEOS_WITH_GTK "Build native GTK client" OFF)
option(REMOTEOS_WITH_BROWSER "Build standalone WebKitGTK browser" OFF)
option(REMOTEOS_WITH_MEDIA "Build GStreamer media transport" OFF)
option(REMOTEOS_BUILD_TESTS "Build tests" ON)
option(REMOTEOS_WITH_DESKTOP_HOST "Build application desktop capture without custom drivers" ${REMOTEOS_WITH_GTK})
option(REMOTEOS_BUILD_HOST_APP "Build the desktop host launcher" ON)
option(REMOTEOS_WITH_SERVICE_HOST "Build the optional privileged host integrations" ON)
set(REMOTEOS_WAYLAND_DEFAULT OFF)
if(REMOTEOS_WITH_DESKTOP_HOST AND REMOTEOS_WITH_MEDIA AND UNIX AND NOT APPLE)
  set(REMOTEOS_WAYLAND_DEFAULT ON)
endif()
option(REMOTEOS_WITH_WAYLAND_HOST "Build portal/PipeWire application hosting on Linux" ${REMOTEOS_WAYLAND_DEFAULT})
find_package(OpenSSL 3.0 REQUIRED)
find_package(Threads REQUIRED)
add_library(remoteos_core src/core/protocol.cpp src/core/identity.cpp src/core/transport.cpp src/core/client_session.cpp src/core/discovery.cpp)
target_include_directories(remoteos_core PUBLIC include third_party third_party/asio/include PRIVATE src/core)
target_compile_definitions(remoteos_core PUBLIC ASIO_STANDALONE ASIO_NO_DEPRECATED)
target_link_libraries(remoteos_core PUBLIC OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
if(WIN32)
  target_compile_definitions(remoteos_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(remoteos_core PUBLIC ws2_32 mswsock advapi32)
endif()
if(MSVC)
  target_compile_options(remoteos_core PRIVATE /W4)
else()
  target_compile_options(remoteos_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

    install(TARGETS remoteos-host-service RUNTIME DESTINATION bin)
  endif()

  set_source_files_properties(src/platform/host_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(remoteos_platform PRIVATE "-framework ScreenCaptureKit" "-framework ApplicationServices" "-framework Carbon" "-framework CoreMedia" "-framework CoreVideo")
else()
  target_sources(remoteos_platform PRIVATE src/platform/host_linux.cpp src/platform/audio_session.cpp)
  target_sources(remoteos_platform PRIVATE src/platform/host_desktop_linux.cpp)

    target_compile_definitions(remoteos_platform PRIVATE REMOTEOS_HAVE_DESKTOP_X11=1)
    target_link_libraries(remoteos_platform PRIVATE PkgConfig::DESKTOP_X11)

  install(TARGETS remoteos-audio-session RUNTIME DESTINATION libexec)

    find_library(EVDI_LIBRARY evdi HINTS "${CMAKE_SOURCE_DIR}/.deps/evdi-1.14.11/library")
  endif()
  if(REMOTEOS_WITH_SERVICE_HOST AND EVDI_INCLUDE_DIR AND EVDI_LIBRARY)
    target_compile_definitions(remoteos_platform PRIVATE REMOTEOS_HAVE_EVDI=1)
    target_include_directories(remoteos_platform PRIVATE ${EVDI_INCLUDE_DIR})
    target_link_libraries(remoteos_platform PRIVATE ${EVDI_LIBRARY})
  endif()

  pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client>=1.20)

  find_program(WAYLAND_SCANNER wayland-scanner REQUIRED)
  enable_language(C)
  set(KDE_SCREENCAST_XML "${CMAKE_SOURCE_DIR}/third_party/kde/zkde-screencast-unstable-v1.xml")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/kde-screencast-client.h"

    COMMAND "${WAYLAND_SCANNER}" private-code "${KDE_SCREENCAST_XML}" "${CMAKE_CURRENT_BINARY_DIR}/kde-screencast-protocol.c"
    DEPENDS "${KDE_SCREENCAST_XML}" VERBATIM)
  target_sources(remoteos_platform PRIVATE src/platform/host_wayland.cpp src/platform/kde_virtual_output.cpp

    if(TARGET remoteos-host-app)
      target_sources(remoteos-host-app PRIVATE packaging/windows/host.manifest)
    endif()

    target_compile_options(remoteos_platform PRIVATE /utf8)
    if(TARGET remoteos-host-app)
      target_compile_options(remoteos-host-app PRIVATE /utf8)
    endif()
  else()
    enable_language(RC)
    configure_file(packaging/windows/host.rc.in ${CMAKE_CURRENT_BINARY_DIR}/remoteos-host.rc @ONLY)
    target_sources(remoteos-host PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/remoteos-host.rc)

      target_sources(remoteos-host-app PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/remoteos-host.rc)
    endif()
  endif()
